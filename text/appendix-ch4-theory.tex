\chapter{\label{app:theory-ch4}Additional Theoretical Results For Chapter 4}

This appendix provides the proofs for Theorems \ref{thrm:4:bts_exp} and \ref{thrm:4:dents_exp}, the exponential convergence results for BTS and DENTS stated in Section \ref{sec:4-5-theory}. The DENTS stochastic process is first recalled in Section \ref{appsec:dents_stoch_process}, and the theoretical results are proven in Section \ref{appsec:exp}.

\section{The DENTS Stochastic Process}
\label{appsec:dents_stoch_process}

    This subsection recalls the \textit{DENTS stochastic process}, with the notation and assumptions from Section \ref{sec:4-5-theory} in-lined.
    
    All value estimates are initialised to zero for all $s\in\cl{S}_0$ and $a\in\cl{A}$:
    \begin{align}
        \mVdents{0}(s) = 0, \quad \mQdents{0}(s,a) = 0, \quad \mHVdents{0}(s) = 0, \quad \mHQdents{0}(s,a) = 0.
    \end{align}

    On the $n$th trial, a trajectory is sampled using the search policy $\tau^n \sim \mpidents{n}$. Where for all $(s,a)\in\cl{S}_0\times\cl{A}$, the search policy is defined as:
    %
    \begin{align}
        \mQdentssft{N^n(s,a)}(s,a) &= \mQdents{N^n(s,a)}(s,a) + \betadents(N^n(s))\mHQdents{N^n(s,a)}(s,a), \label{appeq:dents_q_sft} \\
        \mrhodents{n}(a|s) &= \frac{
                \exp\left(\frac{\mQdentssft{N^n(s,a)}(s,a)}{\alphadents(N^n(s))}\right)
            }{
                \sum\limits_{a'\in\cl{A}} \exp\left(\frac{\mQdentssft{N^n(s,a')}(s,a')}{\alphadents(N^n(s))}\right)
            }.  \label{appeq:full_rho} \\
        \lambda^n(s,\epsdents) &= \min\left(1, \frac{\epsdents}{\log(e+N^{n-1}(s))}\right),  \\
        \mpidents{n}(a|s) &= (1-\mlambdadents{n})\mrhodents{n}(a|s) + \frac{\mlambdadents{n}}{|\cl{A}|}. 
            \label{appeq:dents_search_policy}  \\
    \end{align}

    Letting $\tau^n=(s^n_0,a^n_0,r^n_0,\ldots,s^n_{H-1},a^n_{H-1},r^n_{H-1},s^n_{H})$, the DENTS value and entropy estimates are computed as follows for $t=H-1,\ldots,0$:
    %
    \begin{align}
        \mQdents{N^n(s_t,a_t)}(s_t,a_t) &= 
            R(s_t,a_t) + \sum_{s' \in \suc{s_t}{a_t}} \left( 
                \frac{N^n(s')}{N^n(s_t,a_t)} \mVdents{N^n(s')}(s') \right), 
            \label{appeq:dp_q_backup} \\ 
        \mVdents{N^n(s_t)}(s_t) &=\max_{a\in\cl{A}} \mQdents{N^n(s_t,a)}(s_t,a), 
            \label{appeq:dp_v_backup} \\
        \mHQdents{N^n(s_t,a_t)}(s_t,a_t) &= 
            \sum_{s'\in \suc{s_t}{a_t}} \frac{N^n(s')}{N^n(s_t,a_t)} \mHVdents{N^n(s')}(s'), \\
        \mHVdents{N^n(s_t)}(s_t) &= 
            \cl{H}(\mpidents{n}(\cdot | s_t)) 
                + \sum_{a\in\cl{A}} \mpidents{n}(a_t|s_t)\mHQdents{N^n(s_t,a_t)}(s_t,a_t).
    \end{align}

    Recall that the visit counts are \textit{defined} as a sum of indicator random variables using the sampled trajectories:
    \begin{align}
        N^n(s_t) &= \sum_{i=1}^n \one[s^i_t = s_t], \\
        N^n(s_t,a_t) &= \sum_{i=1}^n \one[s^i_t = s_t \cap a^i_t = a_t].
    \end{align}

    And note that if $s\not\in\tau^n$, then $N^n(s)=N^{n-1}(s)$, and similarly if $(s,a)\not\in\tau^n$, then $N^n(s,a)=N^{n-1}(s,a)$. More specifically, this implies, for example, that if $s\not\in\tau^n$ then:
    \begin{align}
        \mVdents{N^n(s)}(s) = \mVdents{N^{n-1}(s)}(s).
    \end{align}

    Finally, the maximum value and most visited recommendation policies after $n$ trials are respectively defined as:
    \begin{align}
        \mpsidents{n}(s) &= \argmax_{a\in\cl{A}}\mQdents{N^n(s,a)}(s,a), \\
        \mmvdents{n}(s) &= \argmax_{a\in\cl{A}} N^n(s,a).
    \end{align}

\section{Exponential Convergence Results}
\label{appsec:exp}

    This subsection provides the proofs of Theorems \ref{thrm:4:bts_exp} and \ref{thrm:4:dents_exp}. Similar to Section \ref{sec:4-5-theory}, the main results are shown first, providing forward references to the Lemma's used.

    \thrmDENTSexp*
    \begin{proof}
        The proof is broken down into the following steps:
        \begin{itemize}
            \item Lemma \ref{lem:min_prob_dents} shows that under the assumptions that at any state, all actions have a strictly positive probability of being selected;
            \item Lemma \ref{lem:sa_to_s} establishes that, under the assumptions, that concentration inequalities with respect to $N^n(s_t,a_t)$ can be lifted to concentration inequalities with respect to $N^n(s_t)$;
            \item Lemma \ref{lem:s_to_sa} similarly establishes that concentration inequalities with respect to $N^n(s_{t+1})$ can be lifted to inequalities with respect to $N^n(s_t,a_t)$;
            \item Lemmas \ref{lem:stochastic_step} and \ref{lem:dents_val_induction_step} provide the inductive steps used to show concentration inequalities on the value estimates of the DENTS process in Theorem \ref{thrm:dents_val_converge};
            \item Lemma \ref{lem:dents_imm_simple_regret} then uses Theorem \ref{thrm:dents_val_converge} to provide exponential bounds on the immediate simple regret of the DENTS process;
            \item Finally, this theorem holds by using Lemma \ref{lem:imm_regr_to_simp_regr} to lift the immediate simple regret bounds from Lemma \ref{lem:dents_imm_simple_regret} to bounds on the simple regret.
        \end{itemize}
    \end{proof}



    \thrmBTSexp*
    \begin{proof}
        Follows from the proof of Theorem \ref{thrm:4:dents_exp} with $\betadents(x)=0$. 
    \end{proof}


    The bounds shown in this appendix are \textit{problem dependent}, where the constants may depend on the MDP $\cl{M}$.

    \begin{defn}
        \label{def:subopt_gap}
        The \textnormal{suboptimality gap} $\Delta_{\cl{M}}$ of an MDP $\cl{M}$ is smallest difference in optimal values between an optimal action and a suboptimal action, over the reachable states $\cl{S}_0$ (Definition \ref{def:2:reachable}). That is, let $a^*_s = \argmax_a Q^*(s,a)$. The suboptimality gap is then:
        \begin{align}
            \Delta_{\cl{M}} = \min_{s\in\cl{S}_0} \left( Q^*(s,a^*_s) - \max_{a\in\cl{A}-\{a^*_s\}} Q^*(s,a) \right).
        \end{align}
    \end{defn}






\subsection{Lemmas for Exponential Convergence Theorems}
\label{appsec:exp-prelim}

    Firstly, a standard consequence of the union bound is that intersections over exponentially likely events are also exponentially likely (and conversely for unions over exponentially unlikely events) will be useful, and is stated explicitly for reference.
    %
    \begin{lemma} 
    \label{lem:union_bound}
        Let $A_1,...,A_{\ell}$ be some events that each satisfy for $1\leq i \leq \ell$ the inequality $\Pr(\lnot A_i) \leq C_i\exp(-k_i)$, then:
        %
        \begin{align}
            \Pr\left(\bigcup_{i=1}^\ell \lnot A_i\right) &\leq C\exp(-k), \label{appeq:comb_exp_like} \\
            \Pr\left(\bigcap_{i=1}^\ell A_i\right) = 1-\Pr\left(\bigcup_{i=1}^\ell \lnot A_i\right) &\geq 1-C\exp(-k), \label{appeq:exp_likely}
        \end{align}
        %
        where $C=\sum_{i=1}^\ell C_i$ and $k = \min_i k_i$.
    \end{lemma}
    %
    \begin{proofoutline}
        Lemma \ref{lem:union_bound} is a consequence of the union bound, using $\exp(-k_i)\leq\exp(-k)$ and simplifying. Inequality (\ref{appeq:exp_likely}) is just a negation of Inequality (\ref{appeq:comb_exp_like}).
    \end{proofoutline}

    \proofsep




    Additionally, Hoeffding's inequality will be useful to bound the difference between a sum of indicator random variables and its expectation. 
    %
    \begin{theorem} \label{thrm:hoeffding}
        Let $\{X_i\}_{i=1}^k$ be indicator random variables (i.e. $X_i\in\{0,1\}$), and $S_k=\sum_{i=1}^k X_i$. Then Hoeffding's inequality for indicator random variables states for any $\varepsilon > 0$ that:
        \begin{align}
            \Pr(|S_k - \bb{E}S_k|>\varepsilon)\leq 2\exp\left(-\frac{2\varepsilon^2}{k}\right).
        \end{align}
    \end{theorem}
    \begin{proof}
        This is a specific case of Hoeffding's inequality. See \cite{hoeffding} for proof.
    \end{proof}





    When there is some finite $L,U$ such that $\alphadents(x) \geq L > 0$ and $\betadents(x) \leq U < \infty$, then the DENTS process will have a strictly positive probability for all actions in Equation \eqref{appeq:dents_search_policy}. In other words, under these conditions, the DENTS search policy will always have some uniform component to it.
    %
    \begin{lemma} \label{lem:min_prob_dents}
        For any DENTS process with some $L,U$ such that $\alphadents(x) \geq L > 0$ and $\betadents(x) \leq U < \infty$. Then there exists some $\pi^{\min}>0$, for any state $s_t\in\cl{S}_0$, for all $a_t\in\cl{A}$ and any number of trials $n\in\bb{N}$, such that $\mpidents{n}(a_t|s_t)\geq\pi^{\min}$.
    \end{lemma}
    %
    \begin{proofoutline}
        Pick an arbitrary $n\in\bb{N}$, $s\in\cl{S}_0$ and $a\in\cl{A}$. It suffices to show that $\mpidents{n}(a|s) > 0$.

        Let $R_{\min} \leq R(s,a) \leq R_{\max}$ for all $s\in\cl{S}$ and $a\in\cl{A}$. Then the soft Q-values used in DENTS (Equation \ref{appeq:dents_q_sft}) are bounded:
        \begin{align}
            HR_{\min} \leq \mQdentssft{n}(s,a) \leq HR_{\max} + U\log\left(|\cl{A}|\right),
        \end{align}
        %
        using the assumption that $\betadents(x) \leq U < \infty$.

        Because $\alphadents(x) \geq L > 0$, the arguments to all exponentials in the Boltzmann policy (Equation \eqref{appeq:full_rho}) are bounded above and below, so then $\mrhodents{n}(a|s) > 0$.

        Because $\frac{1}{|\cl{A}|} > 0$, the search policy must also be positive $\mpidents{n}(a|s) > 0$, because it is the linear combination of two positive terms (Equation \ref{appeq:dents_search_policy}).

        Finally, because $n,s,a$ were arbitrary, there must be some 
        \begin{align}
            \pi^{\min} = \min_{n\in\bb{N},s\in\cl{S}_0,a\in\cl{A}} \mpidents{n}(a|s) > 0.
        \end{align}
    \end{proofoutline}

    \proofsep














\subsection{Inductive Steps for Exponential Convergence Theorems}
\label{appsec:exp-inductive}

    For the induction, it is convenient to be able to `translate' bounds that depend on some $N^n(s_t,a_t)$ to a corresponding bound on $N^n(s_t)$. In every application below, the event of interest concerns the value estimates in the subtree below $(s_t,a_t)$, and so is determined by the trajectories that pass through $(s_t,a_t)$:
    \begin{lemma} \label{lem:sa_to_s}
        Consider any DENTS process with positive $\alphadents$ and bounded $\betadents$, and some state action pair $(s_t,a_t)\in\cl{S}_0\times\cl{A}$. Suppose that the action $a_t$ has some minimum probability $\eta$ of being chosen from the state $s_t$ (irrespective of the number of trials), i.e. $\Pr(a^i_t=a_t|s^i_t=s_t)\geq\eta$. Let $E$ be an event that is determined by the trajectories passing through $(s_t,a_t)$, so that conditioning further on $N^n(s_t)$ carries no additional information:
        \begin{align}
            \Pr\left(E\ \big|\ N^n(s_t,a_t), N^n(s_t)\right) = \Pr\left(E\ \big|\ N^n(s_t,a_t)\right).
                \label{eq:sa_to_s_cond_indep}
        \end{align}

        And suppose for some $C',k'>0$ that $E$ admits a bound that holds almost surely, conditional on the visit count:
        \begin{align}
            \Pr\left(E\ \big|\ N^n(s_t,a_t)\right) \leq C'\exp(-k'N^n(s_t,a_t)). \label{eq:sa_to_s_assume_bound}
        \end{align}

        Then, there exists $C,k>0$ such that, almost surely:
        \begin{align}
            \Pr\left(E\ \big|\ N^n(s_t)\right) \leq C\exp(-k N^n(s_t)).
        \end{align}
    \end{lemma}
    
    \begin{proof}
        Recall from Equation (\ref{eq:4:nsa_sum}) that 
        \begin{align}
            N^n(s_t,a_t)=\sum_{i\in T^n(s_t)} \one[a^i_t=a_t] = \sum_{i\in T^n(s_t)} \one[a^i_t=a_t | s^i_t=s_t].
        \end{align} 
        
        Taking the expectation over $a_t$ and given $N^n(s_t)$ gives:
        \begin{align}
            \bb{E}\left[N^n(s_t,a_t)\ |\ N^n(s_t)\right] = \sum_{i\in T^n(s_t)} \Pr\left(a^i_t=a_t | s^i_t=s_t\right).
        \end{align}
        
        Using the assumed lower bound $\Pr(a^i_t=a_t|s^i_t=s_t)\geq\eta$, $|T^n(s_t)|=N^n(s_t)$ and rearranging gives:
        \begin{align}
            \bb{E}\left[N^n(s_t,a_t)\ |\ N^n(s_t)\right] - \frac{\eta N^n(s_t)}{2} \geq \frac{\eta N^n(s_t)}{2}. \label{local:d:1}
        \end{align}
        
        So then the probability of $N^n(s_t,a_t)$ being below a multiplicative ratio of $N^n(s_t)$ can be bounded above as follows, where all of the probabilities are conditional on $N^n(s_t)$, so that (\ref{local:d:3}) is a sum of $N^n(s_t)$ indicator random variables:
        \begin{align}
            & \Pr\left(N^n(s_t,a_t) < \frac{1}{2}\eta N^n(s_t)\ \bigg|\ N^n(s_t)\right) \\
                \leq & \Pr\left(N^n(s_t,a_t) < \bb{E}\left[N^n(s_t,a_t)|N^n(s_t)\right] - \frac{1}{2}\eta N^n(s_t)\ \bigg|\ N^n(s_t)\right) \\
                = & \Pr\left(\bb{E}\left[N^n(s_t,a_t)|N^n(s_t)\right] - N^n(s_t,a_t) > \frac{1}{2}\eta N^n(s_t)\ \bigg|\ N^n(s_t)\right)
                    \label{local:d:2} \\
                \leq & \Pr\left(\left|\bb{E}\left[N^n(s_t,a_t)|N^n(s_t)\right] - N^n(s_t,a_t)\right| > \frac{1}{2}\eta N^n(s_t)\ \bigg|\ N^n(s_t)\right)
                    \label{local:d:3} \\
                \leq & 2\exp\left(-\frac{1}{2}\eta^2N^n(s_t)\right). \label{eq:bound_nsa}
        \end{align}
        
        The first inequality follows from Equation \eqref{local:d:1}, the second line is a rearrangement, the third line comes from the inequality in (\ref{local:d:2}) implying the inequality in (\ref{local:d:3}), and the final line uses Theorem \ref{thrm:hoeffding}, a Hoeffding bound for the sum of indicator random variables. 
        
        Finally, the bound using $N^n(s_t,a_t)$ can be converted into one depending on $N^n(s_t)$ by conditioning on the visit counts. Let $E'$ denote the event that $N^n(s_t,a_t) \geq \frac{1}{2}\eta N^n(s_t)$, and note that the indicator $\one[E']$ is determined by $N^n(s_t)$ and $N^n(s_t,a_t)$. Splitting on $E'$ using $\one[E']+\one[\lnot E']=1$ then gives almost surely:
        \begin{align}
             & \Pr\left( E\ \bigg|\ N^n(s_t,a_t) \right) \label{local:d:29} \\
                = & \one[E']\Pr\left( E\ \bigg|\ N^n(s_t,a_t) \right)
                    + \one[\lnot E']\Pr\left( E\ \bigg|\ N^n(s_t,a_t) \right) \\
                \leq & \one[E']\ C'\exp\left(-k' N^n(s_t,a_t)\right) + \one[\lnot E'] \label{local:d:30} \\
                \leq & C'\exp\left(-\frac{1}{2}k'\eta N^n(s_t) \right) + \one[\lnot E']. \label{local:d:31}
        \end{align}

        Inequality \eqref{local:d:30} follows from the assumed Inequality \ref{eq:sa_to_s_assume_bound} and the trivial bound $\Pr\left( E\ |\ N^n(s_t,a_t) \right) \leq 1$. The final line follow by considering the when $E'$ is true or false. When $\one[E']=0$ the inequality is trivial. When $\one[E']=1$, the event $E'$ is true, so $N^n(s_t,a_t) \geq \frac{1}{2}\eta N^n(s_t)$, which gives $C'\exp\left(-k' N^n(s_t,a_t)\right) \leq C'\exp\left(-\frac{1}{2}k'\eta N^n(s_t) \right)$.

        Taking expectations conditional on $N^n(s_t)$, the left hand side \eqref{local:d:29} becomes $\Pr\left( E\ |\ N^n(s_t) \right)$ by the tower property, and $\bb{E}\left[\one[\lnot E']\ |\ N^n(s_t)\right]=\Pr\left(\lnot E'\ |\ N^n(s_t)\right)$, which is precisely the quantity bounded by (\ref{eq:bound_nsa}):
        \begin{align}
            \Pr\left( E\ \bigg|\ N^n(s_t) \right)
                \leq & C'\exp\left(-\frac{1}{2}k'\eta N^n(s_t) \right) +
                    2\exp\left(-\frac{1}{2}\eta^2N^n(s_t)\right) \label{local:d:4} \\
                \leq & C\exp(-k N^n(s_t)). \label{local:d:5}
        \end{align}
        The final inequality (\ref{local:d:5}) follows from Lemma \ref{lem:union_bound}, with $C=C'+2$ and $k = \min(k'\eta/2, \eta^2/2)$.
    \end{proof}

    \proofsep




    

    Similar to Lemma \ref{lem:sa_to_s}, it is convenient to be able to translate bounds that depend on $N^n(s_{t+1})$, for some $s_{t+1}\in\suc{s_t}{a_t}$, into bounds that depend on $N^n(s_t,a_t)$:
    \begin{lemma} \label{lem:s_to_sa}
        Consider any DENTS process with positive $\alphadents$ and bounded $\betadents$. For some state action pair $(s_t,a_t)\in\cl{S}_0\times\cl{A}$, and for some $s'_{t+1}\in\suc{s_t}{a_t}$, let $E$ be an event that is determined by the trajectories passing through $s'_{t+1}$, so that:
        \begin{align}
            \Pr\left(E\ \big|\ N^n(s'_{t+1}), N^n(s_t,a_t)\right) = \Pr\left(E\ \big|\ N^n(s'_{t+1})\right).
                \label{eq:s_to_sa_cond_indep}
        \end{align}
        And suppose for some $C',k'>0$ that $E$ admits a bound that holds almost surely, conditional on the visit count:
        \begin{align}
            \Pr\left(E\ \big|\ N^n(s'_{t+1})\right) \leq C'\exp(-k'N^n(s'_{t+1})).
        \end{align}
        Then, there exists $C,k>0$ such that, almost surely:
        \begin{align}
            \Pr\left(E\ \big|\ N^n(s_t,a_t)\right) \leq C\exp(-k N^n(s_t,a_t)).
        \end{align}
    \end{lemma}
    \begin{proofoutline}
        Proof is similar to Lemma \ref{lem:sa_to_s}. Instead of having a minimum probability of selecting an action $\eta$, replace it with the corresponding probability from the transition distribution $p(s_{t+1}|s_t,a_t)$. Then swapping any $N^n(s_t)$ with $N^n(s_t,a_t)$, any $N^n(s_t,a_t)$ with $N^n(s_{t+1})$, and using $N^n(s_{t+1}) = \sum_{i\in T^n(s_t,a_t)} \one[s^i_{t+1} = s_{t+1}]$ (Equation (\ref{eq:4:ns_sum})) as the sum of indicator random variables will give the result.
    \end{proofoutline}

    \proofsep



    Next, the induction step from $\Vdents$ to $\Qdents$ is given. It is shown that if $\mVdents{N^n(s_{t+1})}(s_{t+1})$ admit exponential concentration bounds, then $\mQdents{N^n(s_t,a_t)}(s_t,a_t)$ must also admit exponential concentration bounds:
    %
    \begin{lemma} \label{lem:stochastic_step}
        Consider any DENTS process with positive $\alphadents$ and bounded $\betadents$, and some state action pair $(s_t,a_t)\in\cl{S}_0\times\cl{A}$. Suppose that for all $s_{t+1}\in\suc{s_t}{a_t}$ there is some $C_{s_{t+1}}, k_{s_{t+1}}>0$ such that for all $\varepsilon_0 >0$:
        \begin{align}
            \Pr\left(\left| \mVdents{N^n(s_{t+1})}(s_{t+1}) - V^*(s_{t+1}) \right| > \varepsilon_0 \right) \leq C_{s_{t+1}}\exp(-k_{s_{t+1}}\varepsilon_0^2 N^n(s_{t+1})).
        \end{align}

        Then there exists some $C,k>0$, for all $\varepsilon>0$ such that:
        \begin{align}
            \Pr\left(\left| \mQdents{N^n(s_t,a_t)}(s_t,a_t) - Q^*(s_t,a_t) \right| > \varepsilon \right) \leq C\exp(-k\varepsilon^2 N^n(s_t,a_t)).
        \end{align}
    \end{lemma}

    \begin{proof}
        By the assumed bounds, Lemma \ref{lem:union_bound} and Lemma \ref{lem:s_to_sa}, there is some $C_1,k_1>0$, such that for any $\varepsilon_1 >0$:
        \begin{align}
            \Pr\left(\forall s_{t+1}. \left|\mVdents{N^n(s_{t+1})}(s_{t+1})-V^*(s_{t+1}) \right| \leq \varepsilon_1 \right) > 1-C_1\exp(-k_1\varepsilon_1^2 N^n(s_t,a_t)). \label{local:one}
        \end{align}

        And recall that for any $\varepsilon_2>0$, using Corollary \ref{cor:4:bound_transition_distribution} that:
        \begin{align}
            & \Pr\left(\max_{s_{t+1}\in\suc{s_t}{a_t}} 
                \left| \frac{N^n(s_{t+1})}{N^n(s_t,a_t)} - p(s_{t+1}|s_t,a_t) \right| 
                \leq \varepsilon_2 \right) \nonumber \\
                    > & 1 - 2 \exp\left(-\frac{1}{2}\varepsilon_2^2 N^n(s_t,a_t) \right). \label{local:two}
        \end{align}
        
        If the events in Inequalities (\ref{local:one}) and (\ref{local:two}) hold, then the following inequalities must also hold:
        \begin{align}
            V^*(s_{t+1})- \varepsilon_1
                &\leq \mVdents{N^n(s_{t+1})}(s_{t+1})
                \leq V^*(s_{t+1})+ \varepsilon_1 \\
            p(s_{t+1}|s_t,a_t) - \varepsilon_2 
                &\leq \frac{N^n(s_{t+1})}{N^n(s_t,a_t)} 
                \leq p(s_{t+1}|s_t,a_t) + \varepsilon_2.
        \end{align} 
        
        The upper bounds on $\mVdents{N^n(s_{t+1})}(s_{t+1})$ and $N^n(s_{t+1})/N^n(s_t,a_t)$ can be used to obtain an upper bound on $\mQdents{N^n(s_t,a_t)}(s_t,a_t)$, starting from the backup from Equation \eqref{appeq:dp_q_backup}:
        \begin{align}
            &\mQdents{N^n(s_t,a_t)}(s_t,a_t) \\
                =& R(s_t,a_t)
                    + \sum_{s_{t+1}\in\suc{s_t}{a_t}}\frac{N^n(s_{t+1})}{N^n(s_t,a_t)}
                        \mVdents{N^n(s_{t+1})}(s_{t+1}) \\
                \leq& R(s_t,a_t)
                    + \sum_{s_{t+1}\in\suc{s_t}{a_t}}\frac{N^n(s_{t+1})}{N^n(s_t,a_t)}
                        \left|\mVdents{N^n(s_{t+1})}(s_{t+1})\right| \\
                \leq& R(s_t,a_t)
                    + \sum_{s_{t+1}\in\suc{s_t}{a_t}}(p(s_{t+1}|s_t,a_t)+\varepsilon_2)
                        (\left|V^*(s_{t+1})\right|+\varepsilon_1) \\
                % =& R(s_t,a_t)
                %     + \bb{E}_{s_{t+1}\sim p(\cdot|s_t,a_t)}[V^*(s_{t+1})]
                %     + \varepsilon_1
                %     + \varepsilon_2\sum_{s_{t+1}\in\suc{s_t}{a_t}} V^*(s_{t+1})
                %     + \varepsilon_1 \varepsilon_2 \\
                \leq& R(s_t,a_t)
                    + \bb{E}_{s_{t+1}\sim p(\cdot|s_t,a_t)}[V^*(s_{t+1})]
                    + \varepsilon_2\left|\sum_{s_{t+1}\in\suc{s_t}{a_t}} V^*(s_{t+1})\right|
                    + \varepsilon_1 + \varepsilon_1 \varepsilon_2 \\
                =& Q^*(s_t,a_t)
                    + \varepsilon_2\left|\sum_{s_{t+1}\in\suc{s_t}{a_t}} V^*(s_{t+1})\right|
                    + \varepsilon_1
                    + \varepsilon_1 \varepsilon_2.
        \end{align}

        Following the similar reasoning but using the lower bounds on $\mVdents{N^n(s_{t+1})}(s_{t+1})$ and $N^n(s_{t+1})/N^n(s_t,a_t)$ gives:
        \begin{align}
            \mQdents{N^n(s_t,a_t)}(s_t,a_t)
                &\geq Q^*(s_t,a_t)
                    - \varepsilon_2\left|\sum_{s_{t+1}\in\suc{s_t}{a_t}} V^*(s_{t+1})\right|
                    - \varepsilon_1
                    - \varepsilon_1 \varepsilon_2.
        \end{align}

        Now given an arbitrary $\varepsilon >0$, recalling that $\varepsilon_1,\varepsilon_2>0$ were arbitrary, set $\varepsilon_1 = \varepsilon/3$ and $\varepsilon_2 = \min(\varepsilon/3, \varepsilon/3|\sum_{s_{t+1}} V^*(s_{t+1})|)$ to give:
        %
        \begin{align}
            Q^*(s_t,a_t) - \varepsilon \leq \mQdents{N^n(s_t,a_t)}(s_t,a_t) \leq Q^*(s_t,a_t) + \varepsilon.
        \end{align}
        
        Using Lemma \ref{lem:union_bound} liberally, there is some $C_2,C_3,k_2,k_3>0$ such that:
        \begin{align}
            &\Pr\left(\left| \mQdents{N^n(s_t,a_t)}(s_t,a_t) - Q^*(s_t,a_t) \right| \leq \varepsilon \right) \\
                >& \left(1-C_1\exp(-k_1\varepsilon_1^2 N^n(s_t,a_t))\right) 
                    \left(1-2\exp\left(-\frac{1}{2}\varepsilon_2^2 N^n(s_t,a_t) \right)\right) \\
                =& \left(1-C_2\exp(-k_2 \varepsilon^2 N^n(s_t,a_t) \right) 
                    \cdot \left(1-C_3\exp(-k_3 \varepsilon^2 N^n(s_t,a_t)\right) \\
                % =& 1 -C_2\exp(-k_2 \varepsilon^2 N^n(s_t,a_t)) -C_3\exp(-k_3 \varepsilon^2 N^n(s_t,a_t)) \notag \\
                %     &+ C_2C_3\exp(-(k_2+k_3)\varepsilon^2 N^n(s_t,a_t) ) \\
                >&\ 1 -C_2\exp(-k_2 \varepsilon^2 N^n(s_t,a_t)) -C_3\exp(-k_3 \varepsilon^2 N^n(s_t,a_t)).
        \end{align}
        
        Finally, by negating, setting $C=C_2+C_3$ and $k=\min(k_2,k_3)$ and using Lemma \ref{lem:union_bound} again, the result follows:
        \begin{align}
                \Pr\left(\left| \mQdents{N^n(s_t,a_t)}(s_t,a_t) - Q^*(s_t,a_t) \right| > \varepsilon \right) \leq C\exp(-k\varepsilon^2 N^n(s_t,a_t)).
        \end{align}
    \end{proof}

    \proofsep








    Lemma \ref{lem:stochastic_step} is now used to show a larger induction step, that if exponential concentration bounds exist for $\mVdents{N^n(s_{t+1})}(s_{t+1})$, then they also exist for $\mVdents{N^n(s_{t})}(s_{t})$:
    %
    \begin{lemma} \label{lem:dents_val_induction_step}
        Consider a DENTS process with positive $\alphadents$ and bounded $\betadents$. Let $s_t\in\cl{S}_0$, with $0\leq t < H$. If for all $s_{t+1}\in\bigcup_{a\in\cl{A}}\suc{s_t}{a}$ there is some $C_{s_{t+1}},k_{s_{t+1}}>0$ such that for any $\varepsilon_{s_{t+1}}>0$:
        \begin{align}
            \Pr\left(\left| \mVdents{N^n(s_{t+1})}(s_{t+1}) - V^*(s_{t+1}) \right| > \varepsilon_{s_{t+1}} \right) 
                &\leq C_{s_{t+1}}\exp\left( -k_{s_{t+1}}\varepsilon_{s_{t+1}}^2 N^n(s_{t+1}) \right), 
        \end{align}
        then there is some $C,k>0$, for any $\varepsilon>0$:
        \begin{align}
            \Pr\left(\left| \mVdents{N^n(s_{t})}(s_{t}) - V^*(s_t) \right| > \varepsilon \right) 
                &\leq C\exp\left( -k\varepsilon^2 N^n(s_{t}) \right).
        \end{align}
    \end{lemma}
    \begin{proof}
        Given the assumptions and by Lemmas \ref{lem:stochastic_step}, \ref{lem:union_bound} and \ref{lem:sa_to_s}, for some $C,k>0$ and for any $\varepsilon_1>0$:
        \begin{align}
            \Pr\left(\forall a_t\in\cl{A}. \left|\mQdents{N^n(s_t,a_t)}(s_t,a_t)-Q^*(s_t,a_t)\right|
                \leq \varepsilon_1\right) &> 1-C \exp(-k \varepsilon_1^2 N^n(s_t)).
        \end{align}
        
        Let $\varepsilon >0$, and set $\varepsilon_1 = \min(\varepsilon,\Delta_{\cl{M}}/2)$ (recalling $\Delta_{\cl{M}}$ from Definition \ref{def:subopt_gap}). So with probability at least $1-C_1 \exp(-k_1 \varepsilon_1^2 N^n(s_t))$ for any $a_t$ it must be that:
        \begin{align}
            Q^*(s_t,a_t) - \varepsilon_1 \leq \mQdents{N^n(s_t,a_t)}(s_t,a_t) &\leq Q^*(s_t,a_t) + \varepsilon_1. 
                \label{local:ten}
        \end{align}
        
        Let $a^*=\max_{a\in\cl{A}} Q^*(s_t,a)$. Using $\varepsilon_1 \leq \Delta_{\cl{M}}/2$ in (\ref{local:ten}), then for any $a\in\cl{A}-\{ a^*\}$:
        \begin{align}
            \mQdents{N^n(s_t,a)}(s_t,a) \leq& Q^*(s_t,a) + \Delta_{\cl{M}}/2 \\
                \leq& Q^*(s_t,a^*) - \Delta_{\cl{M}}/2 \\
                \leq& \mQdents{N^n(s_t,a^*)}(s_t,a^*),
        \end{align}
        and hence $\argmax_{a\in\cl{A}} \mQdents{N^n(s_t,a)}(s_t,a) = a^*$. As a consequence:
        \begin{align}
            \mVdents{N^n(s_t)}(s_t) &= \max_a \mQdents{N^n(s_t,a)}(s_t,a) = \mQdents{N^n(s_t,a^*)}(s_t,a^*). 
                \label{local:editing_one}
        \end{align}
        
        Then using (\ref{local:ten}) with $a_t=a^*$ (noting $V^*(s_t)=Q^*(s_t,a^*)$), using (\ref{local:editing_one}) and using $\varepsilon_1 \leq \varepsilon$ gives:
        \begin{align}
            V^*(s_t) - \varepsilon
                \leq& V^*(s_t) - \varepsilon_1 \\
                \leq& \mVdents{N^n(s_t)}(s_t) \\
                \leq& V^*(s_t) + \varepsilon_1 \\
                \leq& V^*(s_t) + \varepsilon. 
        \end{align}
        
        Hence:
        \begin{align}
            \Pr\left(\left|\mVdents{N^n(s_t)}(s_t)-V^*(s_t)\right| > \varepsilon\right) 
                &\leq C \exp(-k \varepsilon_1^2 N^n(s_t)) \\
                &\leq C \exp(-k \varepsilon^2 N^n(s_t)),
        \end{align}
        which is the result.
    \end{proof}


    \proofsep










    Now the following Theorem performs the induction using Lemma \ref{lem:dents_val_induction_step} to show that the value estimates for all reachable states converge to the optimal values in the DENTS process:
    %
    \begin{theorem} \label{thrm:dents_val_converge}
        Consider a DENTS process with positive $\alphadents$ and bounded $\betadents$, let $s_t\in\cl{S}_0$ then there is some $C,k>0$ for any $\varepsilon>0$:
        \begin{align}
            \Pr\left(\left| \mVdents{N^n(s_{t})}(s_t) - V^*(s_{t}) \right| > \varepsilon \right) 
                &\leq C\exp\left( -k\varepsilon^2 N^n(s_{t}) \right).
        \end{align}
        Moreover, at the root node $s_0$:
        \begin{align}
            \Pr\left(\left| \mVdents{N^n(s_{0})}(s_0) - V^*(s_0) \right| > \varepsilon \right) 
                &\leq C\exp\left( -k\varepsilon^2 n \right). \label{local:four}
        \end{align}
    \end{theorem}
    \begin{proof}
        The result vacuously holds for $t=H$ because $\mVdents{N^n(s_H)}(s_H)=V^*(s_H)=0$. Hence the result holds for all $t=0,...,H$ by induction using Lemma \ref{lem:dents_val_induction_step}. Noting that $N^n(s_0)=n$ gives (\ref{local:four}).
    \end{proof}

    \proofsep



    

    Finally, the concentration inequalities over the value estimates of the DENTS process can then be used to show that the immediate simple regret at all states tends to zero exponentially: 
    %
    \begin{lemma} \label{lem:dents_imm_simple_regret}
        Consider a DENTS process with positive $\alphadents$ and bounded $\betadents$. Let $s_t\in\cl{S}_0$, with $0\leq t \leq H$ then there is some $C',k'>0$ such that:
        \begin{align}
            \bb{E} \ireg(s_{t},\mpsidents{n}) &\leq C'\exp\left( -k' N^n(s_{t}) \right).
        \end{align}
    \end{lemma}
    \begin{proof}
        Let $a^*$ be the locally optimal standard action, so $a^*=\argmax_{a\in\cl{A}} Q^*(s_t,a)$. By Theorem \ref{thrm:dents_val_converge} and Lemmas \ref{lem:stochastic_step} and \ref{lem:sa_to_s} there exists $C_1,k_1>0$ such that for all $\varepsilon_1>0$:
        \begin{align}
            \Pr\left(\forall a_t\in\cl{A}. \left|\mQdents{N^n(s_t,a_t)}(s_t,a_t)-Q^*(s_t,a_t)\right|\leq \varepsilon_1\right) &> 1-C_1 \exp(-k_1 \varepsilon_1^2 N^n(s_t)).
        \end{align}
        
        Setting $\varepsilon_1=\Delta_{\cl{M}}/2$ then gives with probability $1-C_1 \exp(-k_2N^n(s_t))$ (where $k_2=k_1\Delta_{\cl{M}}/2$) for all actions $a\in\cl{A}$ that:
        \begin{align}
            Q^*(s_t,a) - \Delta_{\cl{M}}/2 \leq \mQdents{N^n(s_t,a)}(s_t,a) \leq Q^*(s_t,a) + \Delta_{\cl{M}}/2.
        \end{align}
        
        And hence, with probability $1-C_1 \exp(-k_2N^n(s_t))$, for all $a_t\in\cl{A}-\{a^*\}$ it must be that:
        \begin{align}
            \mQdents{N^n(s_t,a)}(s_t,a_t)
                \leq& Q^*(s_t,a_t) + \Delta_{\cl{M}}/2 \\
                \leq& Q^*(s_t,a^*) - \Delta_{\cl{M}}/2 \\
                \leq& \mQdents{N^n(s_t,a^*)}(s_t,a^*). \label{local:eleven}
        \end{align}
        
        Negating the probability of (\ref{local:eleven}) then gives:
        \begin{align}
            \Pr\left(\mQdents{N^n(s_t,a)}(s_t,a) > \mQdents{N^n(s_t,a^*)}(s_t,a^*)\right) \leq C_1 \exp(-k_2N^n(s_t))
        \end{align}
        
        Finally, the bound on the expected immediate regret follows:
        \begin{align}
            & \bb{E}\ireg(s_t,\mpsidents{n})  \\
                =& \sum_{a\in\cl{A}-\{a^*\}} 
                    \left(V^*(s_t)-Q^*(s_t,a)\right) \Pr\left(\mpsidents{n}(s_t) =a \right) \\
                =& \sum_{a\in\cl{A}-\{a^*\}} 
                    \left(V^*(s_t)-Q^*(s_t,a)\right) \Pr\left(a = \argmax_{a'} \mQdents{N^n(s_t,a)}(s_t,a) \right) \\
                \leq& \sum_{a\in\cl{A}-\{a^*\}} 
                    \left(V^*(s_t)-Q^*(s_t,a)\right) 
                    \Pr\left(\mQdents{N^n(s_t,a)}(s_t,a) > \mQdents{N^n(s_t,a^*)}(s_t,a^*)  \right) \\
                \leq&  \sum_{a\in\cl{A}-\{a^*\}} 
                    \left(V^*(s_t)-Q^*(s_t,a)\right) C_1 \exp(-k_2N^n(s_t)),
        \end{align}
        and setting $k'=k_2$ and $C'=C_1\sum_{a\in\cl{A}-\{a^*\}} \left(V^*(s_t)-Q^*(s_t,a)\right)$ gives the result.
    \end{proof}
